_student_of ||| CREATE OR REPLACE FUNCTION public._student_of(p_token uuid) RETURNS students LANGUAGE sql STABLE SECURITY DEFINER SET search_path TO 'public' AS $function$ select s.* from public.sessions x join public.students s on s.id = x.student_id where x.token = p_token and x.expires_at > now() $function$

admin_list_students ||| CREATE OR REPLACE FUNCTION public.admin_list_students() RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path TO 'public' AS $function$ begin if not public._is_admin() then raise exception 'admin only'; end if; return coalesce((select jsonb_agg(jsonb_build_object('id', s.id, 'name', s.name, 'phone', s.phone, 'level', s.level, 'start_date', s.start_date, 'end_date', s.end_date, 'memo', s.memo, 'days_left', s.end_date - current_date, 'solved', (select count(*) from public.progress p where p.student_id = s.id), 'correct', (select count(*) from public.progress p where p.student_id = s.id and p.correct), 'last_at', (select max(answered_at) from public.progress p where p.student_id = s.id)) order by s.end_date desc, s.name) from public.students s), '[]'::jsonb); end $function$

admin_save_student ||| CREATE OR REPLACE FUNCTION public.admin_save_student(p_id uuid, p_name text, p_phone text, p_level text, p_start date, p_end date, p_memo text) RETURNS uuid LANGUAGE plpgsql SECURITY DEFINER SET search_path TO 'public' AS $function$ declare v uuid; ph text := regexp_replace(coalesce(p_phone, ''), '\D', '', 'g'); begin if not public._is_admin() then raise exception 'admin only'; end if; if length(ph) < 4 then raise exception 'phone'; end if; if p_id is null then insert into public.students (name, phone, level, start_date, end_date, memo) values (btrim(p_name), ph, p_level, coalesce(p_start, current_date), coalesce(p_end, current_date + 90), p_memo) returning id into v; else update public.students set name = btrim(p_name), phone = ph, level = p_level, start_date = p_start, end_date = p_end, memo = p_memo where id = p_id returning id into v; end if; return v; end $function$

student_me ||| CREATE OR REPLACE FUNCTION public.student_me(p_token uuid) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path TO 'public' AS $function$ declare s public.students; begin s := public._student_of(p_token); if s.id is null then return jsonb_build_object('status', 'login'); end if; return jsonb_build_object('status', case when s.end_date >= current_date then 'ok' else 'expired' end, 'profile', public._profile(s)); end $function$
